Nuevas Reglas de Seguridad Firestore

Copia estas reglas en la consola de Firebase para garantizar la máxima seguridad. Permiten que tus pacientes lean tu contenido y te envíen mensajes, pero SOLO TÚ podrás editar la web y leer los mensajes:

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // 1. Contenido Público (Textos, Testimonios, Fotos)
    // Todos pueden leer, solo tú (autenticada) puedes editar
    match /artifacts/{appId}/public/data/{document=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    // 2. Mensajes de Contacto
    // Todos pueden enviar, solo tú puedes leerlos y gestionarlos
    match /artifacts/{appId}/users/{userId}/{document=**} {
      allow create: if true;
      allow read, update, delete: if request.auth != null && request.auth.uid == userId;
    }
  }
}
